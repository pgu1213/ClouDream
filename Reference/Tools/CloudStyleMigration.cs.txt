    public static string Run()
    {
        string[] folders = { "Assets/LostSkiesClouds/Runtime", "Assets/LostSkiesClouds/Editor", "Assets/CloudSea/Runtime", "Assets/CloudSea/Editor" };
        int count = 0;
        foreach (string folder in folders)
        {
            foreach (string path in Directory.GetFiles(folder, "*.cs"))
            {
                if (path.EndsWith("CloudStyleMigration.cs")) { continue; }
                SyntaxNode root = CSharpSyntaxTree.ParseText(File.ReadAllText(path)).GetRoot();
                SyntaxNode changed = new CloudStyleMigration().Visit(root);
                File.WriteAllText(path, changed.NormalizeWhitespace("    ", "\n").ToFullString() + "\n");
                count++;
            }
        }
        return "Formatted " + count + " files";
    }

    private static StatementSyntax Block(StatementSyntax statement)
    {
        if (statement is BlockSyntax) { return statement; }
        return SyntaxFactory.Block(statement);
    }

    public override SyntaxNode VisitIfStatement(IfStatementSyntax node)
    {
        IfStatementSyntax visited = (IfStatementSyntax)base.VisitIfStatement(node);
        return visited.WithStatement(Block(visited.Statement));
    }

    public override SyntaxNode VisitElseClause(ElseClauseSyntax node)
    {
        ElseClauseSyntax visited = (ElseClauseSyntax)base.VisitElseClause(node);
        if (visited.Statement is IfStatementSyntax) { return visited; }
        return visited.WithStatement(Block(visited.Statement));
    }

    public override SyntaxNode VisitForStatement(ForStatementSyntax node)
    {
        ForStatementSyntax visited = (ForStatementSyntax)base.VisitForStatement(node);
        return visited.WithStatement(Block(visited.Statement));
    }

    public override SyntaxNode VisitForEachStatement(ForEachStatementSyntax node)
    {
        ForEachStatementSyntax visited = (ForEachStatementSyntax)base.VisitForEachStatement(node);
        return visited.WithStatement(Block(visited.Statement));
    }

    public override SyntaxNode VisitMethodDeclaration(MethodDeclarationSyntax node)
    {
        MethodDeclarationSyntax visited = (MethodDeclarationSyntax)base.VisitMethodDeclaration(node);
        string description;
        if (!node.GetLeadingTrivia().ToFullString().Contains("summary") && Descriptions.TryGetValue(node.Identifier.Text, out description))
        {
            visited = visited.WithLeadingTrivia(SyntaxFactory.ParseLeadingTrivia("/// <summary>" + description + "</summary>\n"));
        }
        return visited;
    }
